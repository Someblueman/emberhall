The candidate restricted to one *shared* blocking species A admits a useful
possible decomposition. All reactions move exactly one token. Unconditional
reactions form a directed graph R0. Guarded reactions require A to be empty.
On species other than A, form H from all direct edges and every two-step
excursion u→A→v whose second edge is unconditional. Let R0 and H also denote
reflexive transitive closure. A proposed token relation is Q=R0·H·R0, with the
intermediate endpoints outside A. Match initial to final tokens by this relation.
Unconditional-only transport is a separate case.

**Failed first argument:** that is not complete. With only guarded B→A, one B
can reach one A, although Q refuses it. The last guarded step may fill A.
The revised candidate also permits exactly one distinguished token to use
R0·H·(guarded u→A)·R0; every other token uses Q. A possible proof would cut a
valid execution at the first and last guarded firings, reorder unconditional
excursions, and use integral flow for token assignments. That reordering and
its binary-encoding runtime bound have not been proved here. The general
multi-prefix case does not follow from this one-blocker decomposition.

from collections import deque
from itertools import product, permutations
from random import Random

def closure(n, edges):
    r = [[i == j for j in range(n)] for i in range(n)]
    for u,v in edges: r[u][v] = True
    for k in range(n):
        for i in range(n):
            for j in range(n): r[i][j] |= r[i][k] and r[k][j]
    return r

def proposed(n, rules):
    r0 = closure(n, [(u,v) for u,v,g in rules if not g])
    mid = [(u,v) for u,v,g in rules if u and v]
    for u,v,g in rules:
        if u and v == 0:
            mid.extend((u,w) for a,w,h in rules if a == 0 and w and not h)
    h = closure(n, mid)
    q = [[any(r0[i][a] and h[a][b] and r0[b][j]
              for a in range(1,n) for b in range(1,n))
          for j in range(n)] for i in range(n)]
    special = [[any(r0[i][a] and h[a][u] and r0[0][j]
                    for a in range(1,n) for u,v,g in rules if g and v==0)
                for j in range(n)] for i in range(n)]
    return r0, q, special

def transports(a, b, relation):
    origins = tuple(i for i,c in enumerate(a) for _ in range(c))
    ends = tuple(i for i,c in enumerate(b) for _ in range(c))
    return any(all(relation[i][j] for i,j in zip(origins,p))
               for p in set(permutations(ends)))

def reachable(start, rules):
    seen = {start}; pending = deque([start])
    while pending:
        a = pending.popleft()
        for u,v,g in rules:
            if not a[u] or (g and a[0]): continue
            b = list(a); b[u] -= 1; b[v] += 1; b = tuple(b)
            if b not in seen: seen.add(b); pending.append(b)
    return seen

def screen(n, variants, limit):
    pairs = [(u,v) for u in range(n) for v in range(n) if u != v]
    checks = 0
    for choices in variants:
        rules = [(u,v,c==2) for (u,v),c in zip(pairs,choices) if c]
        r0,q,special = proposed(n,rules)
        for mass in range(limit+1):
            states = [x for x in product(range(mass+1),repeat=n) if sum(x)==mass]
            for start in states:
                actual = reachable(start,rules)
                for end in states:
                    guess = transports(start,end,r0) or transports(start,end,q)
                    if not guess:
                        for i in range(n):
                            for j in range(n):
                                if start[i] and end[j] and special[i][j]:
                                    a=list(start); b=list(end); a[i]-=1; b[j]-=1
                                    guess |= transports(a,b,q)
                    checks += 1
                    if guess != (end in actual):
                        raise AssertionError((rules,start,end,guess,sorted(actual)))
    return checks

all_three = product(*([0,1] if u==0 else [0,1,2]
                      for u in range(3) for v in range(3) if u!=v))
print('three species, 324 rule sets, mass <= 3:', screen(3,all_three,3), 'pairs agree')
rng = Random(20261119)
choices4 = [tuple(rng.randrange(2 if u==0 else 3)
            for u in range(4) for v in range(4) if u!=v) for _ in range(300)]
print('four species, 300 fixed-seed rule sets, mass <= 3:',screen(4,choices4,3),'pairs agree')
